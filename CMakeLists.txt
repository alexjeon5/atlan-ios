cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE)
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum iOS version")
  else()
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
  endif()
endif()
project(atlan_ios VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ATLAN_BUILD_IOS "Build the native iOS proof of concept" OFF)
option(ATLAN_BUILD_MACOS "Build the native macOS test application" OFF)
option(ATLAN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(CTest)

if(ATLAN_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_subdirectory(third_party/kernarium)
add_library(atlan_runtime STATIC src/memory.cpp src/runtime.cpp src/demo.cpp src/elf.cpp src/imports.cpp src/lifetime.cpp src/libc_services.cpp src/vfp.cpp src/heap.cpp src/threads.cpp src/files.cpp src/abi.cpp src/android_log.cpp src/jni.cpp src/mmap.cpp src/string_services.cpp src/stdio.cpp src/opensles.cpp src/audio.cpp src/block_cache.cpp src/libc_extra.cpp src/data_symbols.cpp src/text_codec.cpp src/gles.cpp src/gl_recorder.cpp src/glsl.cpp src/gl_context.cpp)
set_source_files_properties(src/vfp.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
target_include_directories(atlan_runtime PUBLIC include)
target_link_libraries(atlan_runtime PRIVATE kernarium_aarch32)
target_compile_options(atlan_runtime PRIVATE -Wall -Wextra -Werror)

add_library(atlan_shell STATIC shell/data_file.cpp shell/data_folder.cpp shell/session.cpp)
target_include_directories(atlan_shell PUBLIC .)
target_link_libraries(atlan_shell PUBLIC atlan_runtime)
target_compile_options(atlan_shell PRIVATE -Wall -Wextra -Werror)

if(APPLE)
  enable_language(OBJCXX)
  add_library(atlan_apple_audio STATIC platform/macos/apple_audio_sink.mm)
  target_include_directories(atlan_apple_audio PUBLIC platform/macos)
  target_link_libraries(atlan_apple_audio PUBLIC atlan_runtime "-framework AudioToolbox" "-framework Foundation")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_link_libraries(atlan_apple_audio PUBLIC "-framework AVFoundation")
  endif()
  target_compile_options(atlan_apple_audio PRIVATE -fobjc-arc -Wall -Wextra -Werror)
endif()

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(atlan-poc tools/poc.cpp)
  target_link_libraries(atlan-poc PRIVATE atlan_runtime)
  add_executable(atlan-elf tools/elf.cpp)
  target_link_libraries(atlan-elf PRIVATE atlan_runtime)
  target_compile_options(atlan-elf PRIVATE -Wall -Wextra -Werror)
  add_executable(atlan-bench bench/bench.cpp)
  target_link_libraries(atlan-bench PRIVATE atlan_runtime)
  target_compile_options(atlan-bench PRIVATE -Wall -Wextra -Werror)
  add_executable(atlan-glsl tools/glsl.cpp)
  target_link_libraries(atlan-glsl PRIVATE atlan_runtime)
  target_compile_options(atlan-glsl PRIVATE -Wall -Wextra -Werror)
  add_executable(atlan-session tools/session.cpp)
  target_link_libraries(atlan-session PRIVATE atlan_shell)
  target_compile_options(atlan-session PRIVATE -Wall -Wextra -Werror)
  install(TARGETS atlan-poc atlan-elf RUNTIME DESTINATION bin)
endif()

if(ATLAN_BUILD_MACOS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "ATLAN_BUILD_MACOS requires a macOS toolchain")
  endif()
  add_subdirectory(platform/macos)
endif()

if(BUILD_TESTING AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(runtime-tests tests/runtime_tests.cpp)
  target_link_libraries(runtime-tests PRIVATE atlan_runtime kernarium_aarch32)
  target_compile_options(runtime-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME runtime-correctness COMMAND runtime-tests)
  add_test(NAME arm32-proof-of-concept COMMAND atlan-poc)
  add_executable(elf-tests tests/elf_tests.cpp)
  target_link_libraries(elf-tests PRIVATE atlan_runtime)
  target_compile_options(elf-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME elf-loader-correctness COMMAND elf-tests)
  add_executable(import-tests tests/import_tests.cpp)
  target_link_libraries(import-tests PRIVATE atlan_runtime)
  target_compile_options(import-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME import-dispatch-correctness COMMAND import-tests)
  add_executable(abi-tests tests/abi_tests.cpp)
  target_link_libraries(abi-tests PRIVATE atlan_runtime)
  target_compile_options(abi-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME abi-jni-correctness COMMAND abi-tests)
  add_executable(service-tests tests/service_tests.cpp)
  target_link_libraries(service-tests PRIVATE atlan_runtime)
  target_compile_options(service-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME host-services-correctness COMMAND service-tests)
  add_executable(text-tests tests/text_tests.cpp)
  target_link_libraries(text-tests PRIVATE atlan_runtime)
  target_compile_options(text-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME text-codec-correctness COMMAND text-tests)
  add_executable(opensles-tests tests/opensles_tests.cpp)
  target_link_libraries(opensles-tests PRIVATE atlan_runtime)
  target_compile_options(opensles-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME opensles-correctness COMMAND opensles-tests)
  add_executable(gles-tests tests/gles_tests.cpp)
  target_link_libraries(gles-tests PRIVATE atlan_runtime)
  target_compile_options(gles-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME gles-correctness COMMAND gles-tests)
  add_executable(glsl-tests tests/glsl_tests.cpp)
  target_link_libraries(glsl-tests PRIVATE atlan_runtime)
  target_compile_options(glsl-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME glsl-translation COMMAND glsl-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/glsl)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    # Compile the MSL the translator emits for the synthetic shaders as C++ against a mock <metal_stdlib> and run it.
    set(GLSL_MOCK_DIR ${CMAKE_CURRENT_BINARY_DIR}/glsl-mock)
    foreach(prog scene flat blend)
      add_custom_command(OUTPUT ${GLSL_MOCK_DIR}/${prog}.metal
        COMMAND ${CMAKE_COMMAND} -E make_directory ${GLSL_MOCK_DIR}
        COMMAND atlan-glsl ${CMAKE_CURRENT_SOURCE_DIR}/tests/glsl/${prog}.vert ${CMAKE_CURRENT_SOURCE_DIR}/tests/glsl/${prog}.frag > ${GLSL_MOCK_DIR}/${prog}.metal
        DEPENDS atlan-glsl ${CMAKE_CURRENT_SOURCE_DIR}/tests/glsl/${prog}.vert ${CMAKE_CURRENT_SOURCE_DIR}/tests/glsl/${prog}.frag
        VERBATIM)
    endforeach()
    add_executable(glsl-mock-tests tests/glsl_mock_tests.cpp ${GLSL_MOCK_DIR}/scene.metal ${GLSL_MOCK_DIR}/flat.metal ${GLSL_MOCK_DIR}/blend.metal)
    set_source_files_properties(${GLSL_MOCK_DIR}/scene.metal ${GLSL_MOCK_DIR}/flat.metal ${GLSL_MOCK_DIR}/blend.metal PROPERTIES HEADER_FILE_ONLY TRUE)
    target_include_directories(glsl-mock-tests PRIVATE tests/mock_metal ${GLSL_MOCK_DIR})
    target_link_libraries(glsl-mock-tests PRIVATE atlan_runtime)
    target_compile_options(glsl-mock-tests PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -Wno-unused-variable -Wno-unused-but-set-variable -Wno-unused-function)
    add_test(NAME glsl-mock-metal COMMAND glsl-mock-tests)
  endif()
  add_executable(gl-context-tests tests/gl_context_tests.cpp)
  target_link_libraries(gl-context-tests PRIVATE atlan_runtime)
  target_compile_options(gl-context-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME gl-context-correctness COMMAND gl-context-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/glsl)
  if(APPLE)
    enable_language(OBJCXX)
    add_executable(metal-render-tests tests/metal_render_tests.mm platform/apple/metal_gl_backend.mm)
    target_include_directories(metal-render-tests PRIVATE platform/apple tests)
    target_compile_definitions(metal-render-tests PRIVATE ATLAN_TEST_GLSL_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/glsl")
    target_link_libraries(metal-render-tests PRIVATE atlan_runtime "-framework Metal" "-framework QuartzCore" "-framework Foundation")
    target_compile_options(metal-render-tests PRIVATE -fobjc-arc -Wall -Wextra -Werror)
    set_target_properties(metal-render-tests PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON OBJCXX_EXTENSIONS OFF)
    add_test(NAME metal-render COMMAND metal-render-tests)
  endif()
  add_executable(thread-tests tests/thread_tests.cpp)
  target_include_directories(thread-tests PRIVATE tests)
  target_link_libraries(thread-tests PRIVATE atlan_runtime)
  target_compile_options(thread-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME guest-threads-correctness COMMAND thread-tests)
  add_executable(vfp-tests tests/vfp_tests.cpp)
  target_link_libraries(vfp-tests PRIVATE atlan_runtime)
  target_compile_options(vfp-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME vfp-correctness COMMAND vfp-tests)
  add_executable(shell-tests tests/shell_tests.cpp)
  target_link_libraries(shell-tests PRIVATE atlan_shell)
  target_compile_options(shell-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME shell-correctness COMMAND shell-tests)
  add_executable(block-cache-tests tests/block_cache_tests.cpp)
  target_link_libraries(block-cache-tests PRIVATE atlan_runtime kernarium_aarch32)
  target_compile_options(block-cache-tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME block-cache-differential COMMAND block-cache-tests)
  if(ATLAN_SANITIZERS)
    # ASan stack-use-after-return fake stacks make this test's tiny hot helpers ~100x slower on arm64 Linux.
    set_tests_properties(block-cache-differential PROPERTIES
      ENVIRONMENT "ASAN_OPTIONS=detect_stack_use_after_return=0" TIMEOUT 600)
  endif()
endif()

if(ATLAN_BUILD_IOS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "ATLAN_BUILD_IOS requires an iOS toolchain")
  endif()
  add_subdirectory(platform/ios)
endif()
