cmake_minimum_required(VERSION 3.22)
if(CMAKE_HOST_APPLE)
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum iOS version")
  else()
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version")
  endif()
endif()
project(atlan_ios VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ATLAN_BUILD_IOS "Build the native iOS proof of concept" OFF)
option(ATLAN_BUILD_MACOS "Build the native macOS test application" OFF)
option(ATLAN_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(CTest)
if(ANDROID)
  # Every static library ends up in the app's shared library (platform/android).
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()

if(ATLAN_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_subdirectory(third_party/kernarium)
add_library(atlan_runtime STATIC engine/src/memory.cpp engine/src/runtime.cpp engine/src/demo.cpp engine/src/elf.cpp engine/src/imports.cpp engine/src/lifetime.cpp engine/src/libc_services.cpp engine/src/vfp.cpp engine/src/heap.cpp engine/src/threads.cpp engine/src/files.cpp engine/src/abi.cpp engine/src/android_log.cpp engine/src/jni.cpp engine/src/mmap.cpp engine/src/string_services.cpp engine/src/stdio.cpp engine/src/opensles.cpp engine/src/audio.cpp engine/src/wav.cpp engine/src/block_cache.cpp engine/src/profiler.cpp engine/src/inflate.cpp engine/src/native_functions.cpp engine/src/native_runtime.cpp engine/src/native_images.cpp engine/src/libc_extra.cpp engine/src/data_symbols.cpp engine/src/text_codec.cpp engine/src/gles.cpp engine/src/gl_recorder.cpp engine/src/glsl.cpp engine/src/gl_context.cpp engine/src/sockets.cpp engine/src/scanf.cpp engine/src/system_services.cpp)
set_source_files_properties(engine/src/vfp.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
target_include_directories(atlan_runtime PUBLIC engine/include)
target_link_libraries(atlan_runtime PRIVATE kernarium_aarch32)
target_compile_options(atlan_runtime PRIVATE -Wall -Wextra -Werror)

# Engine side of the embedding: the guest session and app-neutral host plumbing (docs/engine-adapter.md).
add_library(atlan_shell STATIC engine/shell/data_file.cpp engine/shell/json.cpp engine/shell/sha256.cpp engine/shell/log_file.cpp engine/shell/input.cpp engine/shell/dex.cpp engine/shell/session.cpp engine/shell/http_client.cpp)
target_include_directories(atlan_shell PUBLIC engine .)  # "shell/..." from engine/, "adapters/..." and "platform/..." from the root
target_link_libraries(atlan_shell PUBLIC atlan_runtime)
target_compile_options(atlan_shell PRIVATE -Wall -Wextra -Werror)
# The Atlan app adapter (adapters/atlan): everything that names the app, its files or its Java classes.
add_library(atlan_adapter STATIC adapters/atlan/atlan_adapter.cpp adapters/atlan/cached_lookup.cpp adapters/atlan/charset_conversion.cpp adapters/atlan/data_status.cpp adapters/atlan/glyph_cache.cpp adapters/atlan/poi_lookup.cpp adapters/atlan/quad_geometry.cpp adapters/atlan/speed_readout.cpp adapters/atlan/verified_versions.cpp adapters/atlan/app_package.cpp adapters/atlan/guest_ini.cpp adapters/atlan/java_services.cpp)
target_link_libraries(atlan_adapter PUBLIC atlan_shell)
target_compile_options(atlan_adapter PRIVATE -Wall -Wextra -Werror)

if(APPLE)
  enable_language(OBJCXX)
  add_library(atlan_apple_audio STATIC platform/macos/apple_audio_sink.mm)
  target_include_directories(atlan_apple_audio PUBLIC platform/macos)
  target_link_libraries(atlan_apple_audio PUBLIC atlan_runtime "-framework AudioToolbox" "-framework Foundation")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_link_libraries(atlan_apple_audio PUBLIC "-framework AVFoundation" "-framework UIKit")
  endif()
  target_compile_options(atlan_apple_audio PRIVATE -fobjc-arc -Wall -Wextra -Werror)
endif()

# GlBackend on a real GLES2 (platform/gles): the Android renderer, and on Linux hosts with EGL (Mesa) the same code
# drives headless sessions and tests.
if(ANDROID)
  add_library(atlan_gles STATIC platform/gles/gles_backend.cpp platform/gles/egl_host.cpp)
  target_link_libraries(atlan_gles PUBLIC atlan_runtime EGL GLESv2 android)
elseif(NOT APPLE)
  find_path(ATLAN_EGL_INCLUDE EGL/egl.h)
  find_path(ATLAN_GLES_INCLUDE GLES2/gl2.h)
  find_library(ATLAN_EGL_LIBRARY EGL)
  find_library(ATLAN_GLES_LIBRARY GLESv2)
  if(ATLAN_EGL_INCLUDE AND ATLAN_GLES_INCLUDE AND ATLAN_EGL_LIBRARY AND ATLAN_GLES_LIBRARY)
    add_library(atlan_gles STATIC platform/gles/gles_backend.cpp platform/gles/egl_host.cpp)
    target_include_directories(atlan_gles SYSTEM PUBLIC ${ATLAN_EGL_INCLUDE} ${ATLAN_GLES_INCLUDE})
    target_link_libraries(atlan_gles PUBLIC atlan_runtime ${ATLAN_EGL_LIBRARY} ${ATLAN_GLES_LIBRARY})
  endif()
endif()
if(TARGET atlan_gles)
  target_include_directories(atlan_gles PUBLIC platform/gles)
  target_compile_options(atlan_gles PRIVATE -Wall -Wextra -Werror)
endif()

  add_subdirectory(platform/android)
  add_subdirectory(adapters/atlan/android)

  target_link_libraries(atlan-bench PRIVATE atlan_runtime)
  target_compile_options(atlan-bench PRIVATE -Wall -Wextra -Werror)
  add_executable(atlan-glsl tools/glsl.cpp)
  target_link_libraries(atlan-glsl PRIVATE atlan_runtime)
  target_compile_options(atlan-glsl PRIVATE -Wall -Wextra -Werror)
  add_executable(atlan-verified-entry tools/verified_entry.cpp)
  target_link_libraries(atlan-verified-entry PRIVATE atlan_adapter)
  target_compile_options(atlan-verified-entry PRIVATE -Wall -Wextra -Werror)
  add_executable(atlan-session tools/session.cpp)
  target_link_libraries(atlan-session PRIVATE atlan_adapter)
